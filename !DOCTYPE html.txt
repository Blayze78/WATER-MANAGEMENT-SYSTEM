<script>
let level = 70;
document.getElementById('levelFill').style.width = level + '%';
function checkBill(){
  let sid = document.getElementById('stuId').value;
  let liters = Math.floor(Math.random()*200);
  document.getElementById('billResult').innerText = sid + ' | Used: ' + liters + 'L | Bill: ' + (liters*5) + ' KES';
}
function raiseTicket(){
  let data = {student_id: document.getElementById('stuId').value, type: document.getElementById('type').value, block: document.getElementById('block').value, desc: document.getElementById('desc').value};
  let tickets = JSON.parse(localStorage.getItem('tickets')||'[]');
  tickets.push(data);
  localStorage.setItem('tickets', JSON.stringify(tickets));
  document.getElementById('ticketMsg').innerText = '✅ Ticket raised for ' + data.block;
}
</script>
